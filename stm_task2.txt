	// ------- right motor

	}
}

float decideTargetAngle(char *directionCmd, float *targetAngle) {
// hard-coded angles here

	// front right turn:
	// 90 degrees -> 9.5
	// 180 degrees -> 19.7
	// 270 degrees -> 30.0
	// 360 degrees -> 39.8

	// front left turn
    // 90 degrees -> 9.4
	// 180 degrees -> 19.8
	// 270 degrees -> 30.1
	// 360 degrees -> 40.5

	// backward left turn
    // 90 degrees -> 9.5, 9.6
	// 180 degrees -> 19.9, 20.3
	// 270 degrees -> 30.2, 30.4
	// 360 degrees -> 40.5

	// backward right turn
    // 90 degrees -> 9.5
	// 180 degrees -> 19.9
	// 270 degrees -> 30.2
	// 360 degrees -> 40.6

	if (*directionCmd == 'R') {
		if (*targetAngle == 9.0) return 9.1;
		else if (*targetAngle == 18.0) return 19.5;
		else if (*targetAngle == 27.0) return 29.4;
		else if (*targetAngle == 36.0) return 39.8;
	}

	else if (*directionCmd == 'L') {
		if (*targetAngle == 9.0) return 9.2;
		else if (*targetAngle == 18.0) return 19.3;
		else if (*targetAngle == 27.0) return 29.4;
		else if (*targetAngle == 36.0) return 40.5;
	}

	else if (*directionCmd == 'Q') {
		if (*targetAngle == 9.0) return 9.6;
		else if (*targetAngle == 18.0) return 19.7;
		else if (*targetAngle == 27.0) return 30.2;
		else if (*targetAngle == 36.0) return 40.5;
	}

	else if (*directionCmd == 'E') {
		if (*targetAngle == 9.0) return 9.2;
		else if (*targetAngle == 18.0) return 19.6;
		else if (*targetAngle == 27.0) return 39.9;
		else if (*targetAngle == 36.0) return 40.8;
	}
}

void measureRightIR() {
	uint16_t iDistanceL = 0;
	filtered_irreading = 600; // Initialize filtered IR values variable
	filtered_irreading_int = 0;
	uint16_t caldistance=0;
	uint16_t finaldistance=0;

//	setMotorForward();

	while (1) {
		HAL_ADC_Start(&hadc1);
		HAL_ADC_PollForConversion(&hadc1, 10);
		iDistanceL = HAL_ADC_GetValue(&hadc1 );
		HAL_ADC_Stop(&hadc1); // Ensure ADC1 is stopped before starting ADC2

	  // Low-pass filter equation
	  filtered_irreading = (0.15 * iDistanceL) + ((1 - 0.15) * filtered_irreading);
	  filtered_irreading_int = (int) filtered_irreading;
//	  caldistance= (log((filtered_irreading_int-830.705)/5546.43)/log(0.902366));
//	  finaldistance = (int)caldistance;   very wonky
	  if (filtered_irreading_int >= 600) setMotorForward();
	  else {
		  motorStop();
		  break;
	  }
	}
}

void encoder_task(float *num)
{
  /* USER CODE BEGIN encoder_task */
	if ((*num) <= 30) (*num) = (*num) * 0.5;
	else if ((*num) > 30 && (*num) <= 50) (*num) = (*num) * 0.6;
	else if ((*num) > 50 && (*num) <= 60) (*num) = (*num) * 0.8;
	else if ((*num) > 60 && (*num) <= 80) (*num) = (*num) * 0.85;
	else if ((*num) > 80) (*num) = (*num) * 0.9;

  /* Infinite loop */
  HAL_TIM_Encoder_Start(&htim2,TIM_CHANNEL_ALL); //activate the encoder for Motor A
  HAL_TIM_Encoder_Start(&htim3,TIM_CHANNEL_ALL); //activate the encoder for Motor B
  int left_prev, left_curr, left_diff, right_prev, right_curr, right_diff;
  int dir_L, dir_R;
  int left_speed, right_speed;
  double LEFTWHEEL_DIST = 0;
  double RIGHTWHEEL_DIST = 0;
  char buf[20]; // Buffer to hold the string representation
  uint32_t tick, cur_tick, T;

  left_prev = __HAL_TIM_GET_COUNTER(&htim2);
  right_prev = __HAL_TIM_GET_COUNTER(&htim3);
  tick = HAL_GetTick(); //tick value in milliseconds


  while(LEFTWHEEL_DIST < (*num))
  {
	  cur_tick = HAL_GetTick();
	  if (cur_tick - tick > 50L){ //every 0.005 second
		  left_curr = __HAL_TIM_GET_COUNTER(&htim2);
		  right_curr = __HAL_TIM_GET_COUNTER(&htim3);
		  //Left encoder
		  if(__HAL_TIM_IS_TIM_COUNTING_DOWN(&htim2)){
			  if(left_curr <= left_prev){
				  left_diff = left_prev - left_curr;
			  }
			  else {
				  left_diff = (65535 - left_curr) + left_prev; //handle overflow situation
			  }
			  dir_L = 1;
		  }
		  else {
			  if(left_curr >= left_prev){
				  left_diff = left_curr - left_prev;
			  }
			  else {
				  left_diff = (65535 - left_prev) + left_curr;
			  }
			  dir_L = -1;
		  }
		  //Right encoder
		  if(__HAL_TIM_IS_TIM_COUNTING_DOWN(&htim3)){
			  if(right_curr <= right_prev){
				  right_diff = right_prev - right_curr;
			  }
			  else {
				  right_diff = (65535 - right_curr) + right_prev; //handle overflow situation
			  }
			  dir_R = -1;
		  }
		  else {
			  if(right_curr >= right_prev){
				  right_diff = right_curr - right_prev;
			  }
			  else {
				  right_diff = (65535 - right_prev) + right_curr;
			  }
			  dir_R = 1;
		  }
		  T = cur_tick - tick;
		  //Show speed in ticks/s
		  left_speed = left_diff * (1000/T);
		  right_speed = right_diff * (1000/T);

		  //Calculate distance traveled
//		  distInt_L += left_diff * dir_L;
//		  distInt_R += right_diff * dir_R;
//		  sprintf(OLED_Row_1, "DST L: %6d\0", distInt_L);
//		  sprintf(OLED_Row_2, "DST R: %6d\0", distInt_R);
		  LEFTWHEEL_DIST += (double)left_diff * (20.4/520);
		  RIGHTWHEEL_DIST += (double)right_diff * (20.4/520);

//		   control logic for deciding when to stop depending on input command
		  if (LEFTWHEEL_DIST >= (*num)) {
			  motorStop();
			  break;
		  }

		  sprintf(buf, "Enc. Dist = %d",(int) LEFTWHEEL_DIST);
		  OLED_ShowString(10, 40, buf);
//		  sprintf(buf, "ldist: %d", (int) LEFTWHEEL_DIST);
//		  OLED_ShowString(10, 40, buf);
//		  sprintf(buf, "rdist: %d", (int) RIGHTWHEEL_DIST);
//		  OLED_ShowString(10, 50, buf);
		  OLED_Refresh_Gram();

		  //Reset counters
		  left_prev = __HAL_TIM_GET_COUNTER(&htim2);
		  right_prev = __HAL_TIM_GET_COUNTER(&htim3);
		  tick = HAL_GetTick(); //tick value in milliseconds
//		  OLED_ShowString(60, 0, "reset");

	  }
	  osDelay(20);
  }
  /* USER CODE END encoder_task */
}

void robotTurn(float *targetAngle) {
	char buf[20]; // Buffer to hold the string representation
//	targetAngle = num;
	angleNow = 0;
	gyroZ = 0;

	last_curTask_tick = HAL_GetTick();
	do {
		if (HAL_GetTick() - last_curTask_tick >= 10) { // sample gyro every 10ms
			__Gyro_Read_Z(&hi2c1, readGyroZData, gyroZ);
			angleNow += gyroZ / GRYO_SENSITIVITY_SCALE_FACTOR_2000DPS * 0.01;
			if (fabs(angleNow) >= *targetAngle)
			{
				resetGyro();
				break;
			}
			last_curTask_tick = HAL_GetTick();
			  sprintf(buf, "Angle = %d",(int) angleNow);
			  OLED_ShowString(10, 50, buf);
			  OLED_Refresh_Gram();

		}
	} while (1);

	// After the turn, reset the gyro again to prevent drift accumulation
	resetGyro();
}

void HAL_TIM_IC_CaptureCallback(TIM_HandleTypeDef *htim){

	if(htim==&htim1){
		if(HAL_GPIO_ReadPin(GPIOE, GPIO_PIN_14) == GPIO_PIN_SET){	//If pin on high, means positive edge
			tc1 = HAL_TIM_ReadCapturedValue(htim, TIM_CHANNEL_4);	//Retrive value and store in tc1
		} else if (HAL_GPIO_ReadPin(GPIOE, GPIO_PIN_14) == GPIO_PIN_RESET){	//If pin on low means negative edge
			tc2 = HAL_TIM_ReadCapturedValue(htim, TIM_CHANNEL_4);	//Retrive val and store in tc2
			if (tc2 > tc1){
				echo = tc2-tc1;		//Calculate the differnce = width of pulse
			} else {
				echo = (65536-tc1)+tc2;
			}
		}

	}
}

			aRxBuffer[2] = 'n';

			recentreFromRight45();

			aRxBuffer[1] = 'n';
			aRxBuffer[2] = 'n';
			aRxBuffer[3] = 'n';
			break;
		}

			turnLeft90();
			measureRightIR();

			turnRight180();
			setMotorForward();
			num = 30;
			encoder_task(&num);

			num = total_distance_travelled + 20;
			encoder_task(&num);

			turnRight90();

			// head back to carpark
//			HAL_Delay(500);
			measureLeftIRToPark();
			moveForward10();

			turnLeft90();
			moveIntoCarpark();
			curDirectionCmd = 'n';
			aRxBuffer[0] = 'n';
			aRxBuffer[1] = 'n';
			aRxBuffer[2] = 'n';
			aRxBuffer[3] = 'n';
			break;
		}

		else if (curDirectionCmd == 'E') { // second obstacle encounter right
			turnRight90();

//			HAL_Delay(500);
